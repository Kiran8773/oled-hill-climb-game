struct Veh {
  const char* name;
  float mass, inertia;
  float wrR, wrF;            // rear / front tyre radius
  float dx;                  // half wheelbase
  float dyR, dyF;            // wheel centre below car centre (rear / front)
  float k, c;                // suspension spring / damper per wheel
  float engine, vmax, brake;
  float airTq;               // rotation torque in the air
  float lift;                // 0..1 aerodynamic lift
  uint8_t awd;               // 1 = all wheel drive
  uint8_t shape;             // 0 jeep, 1 monster, 2 buggy, 3 tractor, 4 skyracer, 5 hippi van, 6 tank, 7 supercar
  float bw, by0, by1;        // body half length, lower body y-range
  float cabL, cabR, cabTop;  // cabin
};

// ---------------- sound ----------------
void buzzInit() {
#if defined(ESP_ARDUINO_VERSION_MAJOR) && ESP_ARDUINO_VERSION_MAJOR >= 3
  ledcAttach(PIN_BUZ, 2000, 8);
#else
  ledcSetup(0, 2000, 8);
  ledcAttachPin(PIN_BUZ, 0);
#endif
}
void buzz(int f) {
  if (!soundOn) f = 0;
  if (f == lastFreq) return;
  lastFreq = f;
#if defined(ESP_ARDUINO_VERSION_MAJOR) && ESP_ARDUINO_VERSION_MAJOR >= 3
  ledcWriteTone(PIN_BUZ, f);
#else
  ledcWriteTone(0, f);
#endif
}
void sfx(int f, int ms) { buzz(f); sfxEnd = millis() + ms; }
void melody(const int* n, const int* d, int cnt) {
  if (!soundOn) return;
  for (int i = 0; i < cnt; i++) { buzz(n[i]); delay(d[i]); }
  buzz(0);
}

// ---------------- helpers ----------------
uint32_t hash32(uint32_t v) {
  v ^= v >> 16; v *= 0x7feb352d; v ^= v >> 15; v *= 0x846ca68b; v ^= v >> 16;
  return v;
}

float rampH(float x) {
  float s = floorf((x - 500.0f) / 700.0f);
  if (s < 0) return 0;
  if (s > 0 && hash32((uint32_t)s + 777) % 3 == 0) return 0;
  float u = x - (500.0f + s * 700.0f);
  if (u < 0 || u >= 120.0f) return 0;
  if (u < 80.0f) return 0.3f * u;
  float t = (u - 80.0f) / 40.0f;
  return 12.0f * (1.0f + cosf(t * 3.14159f));
}

// Highly accurate terrain engine generating custom stages
float terrain(float x) {
  float r = (x - 120.0f) / 400.0f;
  if (r < 0) r = 0; if (r > 1) r = 1;
  r = r * r * (3.0f - 2.0f * r);
  float d = x / 9000.0f; if (d > 1) d = 1;

  if (stageIdx == 1) { 
    // MOUNTAIN: A-shaped steep bumps, but perfectly smoothed off at the tips
    float amp = r * (22.0f + 25.0f * d);
    float s1 = sinf(x * 0.008f);
    // Hyperbolic smoothing logic: flattens the sharp tip of a triangle wave
    float a_shape = 1.0f - sqrtf(s1 * s1 + 0.05f); 
    float shape = a_shape * 1.8f + 0.3f * sinf(x * 0.03f) + 0.15f * sinf(x * 0.07f);
    return 20.0f + amp * shape;
  } 
  else if (stageIdx == 2) { 
    // HIGHWAY: Massive amplitude, extremely wide frequency, cubic sine terraces
    float amp = r * (70.0f + 80.0f * d); // Massive vertical jumps
    float s1 = sinf(x * 0.0012f); // Extremely wide distance
    // Cubing a sine wave (s1*s1*s1) creates long flat plateaus, then very steep ramps
    float shape = (s1 * s1 * s1) * 1.4f + 0.25f * sinf(x * 0.0031f) + 0.1f * sinf(x * 0.008f);
    return 20.0f + amp * shape; 
  } 
  else { 
    // STAGE 0: COUNTRYSIDE: The original bouncy hills and ramps
    float amp = r * (4.0f + 11.0f * d);
    float shape = sinf(x * 0.022f) + 0.5f * sinf(x * 0.051f + 1.3f) + 0.22f * sinf(x * 0.13f + 0.6f);
    return 20.0f + amp * shape + rampH(x);
  }
}

bool inRing(int32_t* r, int n, int32_t v) {
  for (int i = 0; i < n; i++) if (r[i] == v) return true;
  return false;
}

bool coinAt(int i, float &wx, float &wy) {
  if (i < 0) return false;
  uint32_t h = hash32(i / 5);
  if (h % 3 == 0) return false;
  wx = 160.0f + i * 22.0f;
  wy = terrain(wx) + 12.0f + (((h >> 4) & 3) == 0 ? 14.0f : 0.0f);
  return true;
}
bool fuelAt(int seg, float &wx, float &wy) {
  if (seg < 0) return false;
  uint32_t h = hash32(seg + 9999);
  if (h % 4 == 0) return false;
  wx = 380.0f + seg * 600.0f + ((h >> 8) % 200);
  wy = terrain(wx) + 12.0f + (((h >> 3) & 1) ? 10.0f : 0.0f);
  return true;
}

inline int SX(float wx) { return (int)(wx - camX + 0.5f); }
inline int SY(float wy) { return (int)(34.5f - (wy - camY)); }

void bodyPoint(float lx, float ly, float &wx, float &wy) {
  float ca = cosf(car.a), sa = sinf(car.a);
  wx = car.x + lx * ca - ly * sa;
  wy = car.y + lx * sa + ly * ca;
}

// ---------------- input ----------------
void readInputs() {
  int rx = analogRead(PIN_VRX) - cxCal;
  int ry = analogRead(PIN_VRY) - cyCal;
  if (INVERT_X) rx = -rx;
  if (INVERT_Y) ry = -ry;

  gasIn = 0; brk = 0;
  if (rx > STICK_DEAD)       gasIn = fminf(1.0f, (rx - STICK_DEAD) / 900.0f);
  else if (rx < -STICK_DEAD) brk   = fminf(1.0f, (-rx - STICK_DEAD) / 900.0f);

  navX = 0; navY = 0;
  unsigned long t = millis();
  if (abs(rx) < 500 && abs(ry) < 500) lastNav = 0;
  if (t - lastNav > 230) {
    if (ry < -1000)      { navY = -1; lastNav = t; }
    else if (ry > 1000)  { navY = 1;  lastNav = t; }
    else if (rx > 1000)  { navX = 1;  lastNav = t; }
    else if (rx < -1000) { navX = -1; lastNav = t; }
  }
}

bool buttonEdge() {
  static bool last = false;
  static unsigned long lastT = 0;
  bool cur = digitalRead(PIN_SW) == LOW;
  bool e = false;
  if (cur && !last && millis() - lastT > 200) { e = true; lastT = millis(); }
  last = cur;
  return e;
}

// ---------------- game logic ----------------
void resetGame() {
  car.x = 20; car.y = terrain(20) + V->wrR + V->dyR + 1;
  car.vx = car.vy = car.a = car.w = car.wa = 0;
  fuel = 100; coins = 0; distM = 0; stuckT = 0;
  airTimer = 0; wasAir = false; flipTracker = 0; popupBonus = 0;
  for (int i = 0; i < RING; i++) coinRing[i] = -1;
  for (int i = 0; i < 8; i++) fuelRing[i] = -1;
  camX = car.x - 36; camY = car.y;
}

void physicsStep(float dt) {
  float ca = cosf(car.a), sa = sinf(car.a);
  float Fx = 0, Fy = 0, T = 0;
  int contacts = 0;
  float g = fuel > 0 ? gasIn : 0;

  for (int i = 0; i < 2; i++) {
    float lx = (i == 0) ? -V->dx : V->dx;
    float ly = (i == 0) ? -V->dyR : -V->dyF;
    float wr = (i == 0) ? V->wrR : V->wrF;
    float rx = lx * ca - ly * sa;
    float ry = lx * sa + ly * ca;
    float wx = car.x + rx, wy = car.y + ry;

    float gh = terrain(wx);
    float s = (terrain(wx + 1.0f) - terrain(wx - 1.0f)) * 0.5f;
    float il = 1.0f / sqrtf(1.0f + s * s);
    float nx = -s * il, ny = il, tx = il, ty = s * il;

    float pen = gh + wr - wy;
    if (pen > -1.0f) contacts++;
    if (pen <= 0) continue;

    float wvx = car.vx - car.w * ry;
    float wvy = car.vy + car.w * rx;
    float vn = wvx * nx + wvy * ny;
    float vt = wvx * tx + wvy * ty;

    float Fn = V->k * pen - V->c * vn;
    if (Fn < 0) Fn = 0;
    if (Fn > 1500.0f * V->mass) Fn = 1500.0f * V->mass;

    // Bounce bonus trigger
    if (Fn > 300.0f * V->mass && fabsf(vt) > 20.0f) {
      if (millis() > sfxEnd) {
        coins += 1; 
        sfx(1600, 30);
      }
    }

    float Ft = 0;
    bool driven = (i == 0) || V->awd;
    if (g > 0 && driven && vt < V->vmax) {
      float share = V->awd ? 0.5f : 1.0f;
      Ft += V->engine * share * g * (1.0f - vt / V->vmax);
    }
    if (brk > 0) {
      if (vt > 4.0f) Ft -= V->brake * brk;
      else if (i == 0 && vt > -60.0f) Ft -= V->engine * 0.4f * brk;
    }
    Ft -= vt * ROLL_RES * V->mass;

    float fx = Fn * nx + Ft * tx;
    float fy = Fn * ny + Ft * ty;
    Fx += fx; Fy += fy;
    T += rx * fy - ry * fx;
  }

  float ax = Fx / V->mass - car.vx * AIR_DRAG;
  float ay = -GRAV + Fy / V->mass;
  float aw = T / V->inertia - car.w * ANG_DAMP;

  if (contacts == 0) {
    if (!wasAir) {
      wasAir = true;
      airTimer = 0;
      flipTracker = 0; 
    }
    airTimer += dt;
    flipTracker += car.w * dt;

    if (fabsf(flipTracker) >= 6.28318f) {
      if (flipTracker > 0) flipTracker -= 6.28318f;
      else flipTracker += 6.28318f;

      coins += 50;
      popupBonus = 50;
      popupStr = "FLIP!";
      popupEnd = millis() + 1200;
      if (millis() > sfxEnd) sfx(2000, 100);
    }

    aw += (g - brk) * V->airTq;
    float sp = fabsf(car.vx);
    float L = (sp - LIFT_V0) / LIFT_VR;
    if (L < 0) L = 0; if (L > 1) L = 1;
    L *= V->lift * (1.0f + 1.2f * sinf(car.a));
    if (L < 0) L = 0; if (L > 1.1f) L = 1.1f;
    ay += GRAV * 0.92f * L;

    float th = g * V->engine * AIR_THRUST / V->mass;
    ax += th * ca;
    ay += th * sa;

    float alt = car.y - terrain(car.x);
    if (alt > MAX_ALT) ay -= (alt - MAX_ALT) * 3.0f;
  } else {
    if (wasAir && airTimer > 0.4f) {
      int airBonus = (int)(airTimer * 25.0f);
      coins += airBonus;
      popupBonus = airBonus;
      popupStr = "AIR!";
      popupEnd = millis() + 1200;
      sfx(2500, 100);
    }
    airTimer = 0;
    wasAir = false;
  }

  car.vx += ax * dt; car.vy += ay * dt; car.w += aw * dt;
  if (car.w > 12) car.w = 12; if (car.w < -12) car.w = -12;
  car.x += car.vx * dt;
  car.y += car.vy * dt;
  car.a += car.w * dt;
  car.wa += ((car.vx * ca + car.vy * sa) / V->wrR) * dt;

  if (car.x < 12) { car.x = 12; if (car.vx < 0) car.vx = 0; }
}

void headPos(float &hx, float &hy) {
  hx = (V->cabL + V->cabR) * 0.5f;
  hy = V->by1 + (V->cabTop - V->by1) * 0.55f;
}

bool crashed() {
  float hx, hy; headPos(hx, hy);
  const float pts[3][2] = { {V->cabL + 3, V->cabTop}, {V->cabR - 3, V->cabTop}, {hx, hy} };
  for (int i = 0; i < 3; i++) {
    float px, py; bodyPoint(pts[i][0], pts[i][1], px, py);
    if (py < terrain(px) + 1.0f) return true;
  }
  return false;
}

void pickups() {
  int c0 = (int)floorf((car.x - 160.0f) / 22.0f);
  for (int i = c0 - 1; i <= c0 + 1; i++) {
    float wx, wy;
    if (!coinAt(i, wx, wy) || inRing(coinRing, RING, i)) continue;
    float dx = wx - car.x, dy = wy - car.y;
    if (dx * dx + dy * dy < 160) {
      coinRing[coinHead++ % RING] = i;
      coins++;
      sfx(1900, 40);
    }
  }
  int s0 = (int)floorf((car.x - 380.0f) / 600.0f);
  for (int s = s0 - 1; s <= s0 + 1; s++) {
    float wx, wy;
    if (!fuelAt(s, wx, wy) || inRing(fuelRing, 8, s)) continue;
    float dx = wx - car.x, dy = wy - car.y;
    if (dx * dx + dy * dy < 220) {
      fuelRing[fuelHead++ % 8] = s;
      fuel = fminf(100.0f, fuel + 40.0f);
      sfx(1200, 60);
    }
  }
}

void gameOver(const char* why) {
  overReason = why;
  gameActive = false;
  buzz(0);
  uint32_t m = (uint32_t)distM;
  if (m > bestDist[stageIdx]) { 
    bestDist[stageIdx] = m; 
    char key[4]; snprintf(key, sizeof(key), "b%d", stageIdx);
    prefs.putUInt(key, bestDist[stageIdx]); 
  }
  if (why[0] == 'C') {
    const int n[] = {500, 400, 300, 200};
    const int d[] = {120, 120, 120, 250};
    melody(n, d, 4);
  } else {
    const int n[] = {600, 450, 300};
    const int d[] = {150, 150, 300};
    melody(n, d, 3);
  }
  state = ST_OVER; menuSel = 0;
}

void startRun() {
  resetGame();
  gameActive = true;
  lastRunVeh = vehIdx;
  lastRunStage = stageIdx;
  const int n[] = {800, 1000, 1300};
  const int d[] = {80, 80, 140};
  melody(n, d, 3);
  state = ST_PLAY;
  lastUs = micros();
}

// ---------------- drawing ----------------
void drawTerrain() {
  int prev = 0;
  for (int sx = 0; sx < SCREEN_W; sx++) {
    int gy = SY(terrain(camX + sx));
    if (sx == 0) prev = gy;
    int px = sx > 0 ? sx - 1 : 0;
    display.drawLine(px, prev, sx, gy, WHITE);
    display.drawLine(px, prev + 1, sx, gy + 1, WHITE);
    for (int y = gy + 3; y < SCREEN_H; y += 3)
      if (y >= 0 && ((sx + y) & 3) == 0) display.drawPixel(sx, y, WHITE);
    prev = gy;
  }
}

void drawItems() {
  int c0 = (int)floorf((camX - 160.0f) / 22.0f) - 1;
  for (int i = c0; i < c0 + 9; i++) {
    float wx, wy;
    if (!coinAt(i, wx, wy) || inRing(coinRing, RING, i)) continue;
    int x = SX(wx), y = SY(wy);
    if (x < -4 || x > SCREEN_W + 4) continue;
    display.drawCircle(x, y, 2, WHITE);
    display.drawPixel(x, y, WHITE);
  }
  int s0 = (int)floorf((camX - 380.0f) / 600.0f) - 1;
  for (int s = s0; s < s0 + 3; s++) {
    float wx, wy;
    if (!fuelAt(s, wx, wy) || inRing(fuelRing, 8, s)) continue;
    int x = SX(wx), y = SY(wy);
    if (x < -6 || x > SCREEN_W + 6) continue;
    display.drawRect(x - 3, y - 4, 7, 8, WHITE);
    display.drawFastHLine(x - 1, y, 3, WHITE);
    display.drawFastVLine(x, y - 2, 5, WHITE);
  }
}

void drawWheelAt(int x, int y, int r, float wa) {
  display.fillCircle(x, y, r + 1, BLACK);
  display.drawCircle(x, y, r, WHITE);
  if (r >= 6) display.drawCircle(x, y, r - 1, WHITE);
  if (r >= 8) display.drawCircle(x, y, r - 3, WHITE);
  if (r >= 7) display.drawCircle(x, y, 2, WHITE); else display.drawPixel(x, y, WHITE);
  int l = r - 2;
  int dxs = (int)lroundf(l * cosf(wa)), dys = (int)lroundf(l * sinf(wa));
  display.drawLine(x - dxs, y - dys, x + dxs, y + dys, WHITE);
  if (r >= 7) display.drawLine(x - dys, y + dxs, x + dys, y - dxs, WHITE);
}

void drawVehicle(const Veh* v, float cxs, float cys, float ang, float wa) {
  float ca = cosf(ang), sa = sinf(ang);
  auto P = [&](float lx, float ly, int &x, int &y) {
    x = (int)lroundf(cxs + lx * ca - ly * sa);
    y = (int)lroundf(cys - (lx * sa + ly * ca));
  };
  int ax, ay, bx, by, cx, cy, dx, dy;
  P(-v->bw, v->by0, ax, ay); P(v->bw, v->by0, bx, by);
  P(v->bw, v->by1, cx, cy);  P(-v->bw, v->by1, dx, dy);
  display.fillTriangle(ax, ay, bx, by, cx, cy, WHITE);
  display.fillTriangle(ax, ay, cx, cy, dx, dy, WHITE);

  // cabin
  int c0x, c0y, c1x, c1y, c2x, c2y, c3x, c3y;
  P(v->cabL, v->by1, c0x, c0y);       P(v->cabL + 3, v->cabTop, c1x, c1y);
  P(v->cabR - 3, v->cabTop, c2x, c2y); P(v->cabR, v->by1, c3x, c3y);
  display.drawLine(c0x, c0y, c1x, c1y, WHITE);
  display.drawLine(c1x, c1y, c2x, c2y, WHITE);
  display.drawLine(c2x, c2y, c3x, c3y, WHITE);

  int x1, y1, x2, y2, x3, y3;
  if (v->shape == 1) {            // monster
    P(v->cabL + 3, v->cabTop + 2, x1, y1); P(v->cabR - 3, v->cabTop + 2, x2, y2);
    display.drawLine(x1, y1, x2, y2, WHITE);
    P(v->bw + 2, v->by0, x1, y1); P(v->bw + 2, v->by1, x2, y2);
    display.drawLine(x1, y1, x2, y2, WHITE);
  } else if (v->shape == 3) {     // tractor
    P(v->bw - 3, v->by1, x1, y1); P(v->bw - 3, v->by1 + 5, x2, y2);
    display.drawLine(x1, y1, x2, y2, WHITE);
  } else if (v->shape == 4) {     // skyracer
    P(v->bw, v->by0, x1, y1); P(v->bw, v->by1, x2, y2); P(v->bw + 5, v->by0, x3, y3);
    display.fillTriangle(x1, y1, x2, y2, x3, y3, WHITE);
    P(-v->bw + 1, v->by1, x1, y1); P(-v->bw + 1, v->by1 + 5, x2, y2);
    display.drawLine(x1, y1, x2, y2, WHITE);
    P(-v->bw - 3, v->by1 + 5, x1, y1); P(-v->bw + 6, v->by1 + 5, x2, y2);
    display.drawLine(x1, y1, x2, y2, WHITE);
  } else if (v->shape == 5) {     // Hippi Van
    P(-v->bw + 2, v->cabTop - 1, x1, y1); P(v->bw - 2, v->cabTop - 1, x2, y2);
    display.drawLine(x1, y1, x2, y2, WHITE);
  } else if (v->shape == 6) {     // Military Tank
    P(0, v->cabTop, x1, y1); P(v->bw + 6, v->cabTop - 1, x2, y2);
    display.drawLine(x1, y1, x2, y2, WHITE);
    display.drawLine(x1, y1 - 1, x2, y2 - 1, WHITE);
  } else if (v->shape == 7) {     // Supercar
    P(-v->bw + 2, v->by1, x1, y1); P(-v->bw, v->by1 + 4, x2, y2);
    display.drawLine(x1, y1, x2, y2, WHITE); 
    P(-v->bw - 2, v->by1 + 4, x1, y1); P(-v->bw + 3, v->by1 + 4, x2, y2);
    display.drawLine(x1, y1, x2, y2, WHITE); 
  }

  float hx = (v->cabL + v->cabR) * 0.5f;
  float hy = v->by1 + (v->cabTop - v->by1) * 0.55f;
  P(hx, hy, x1, y1);
  display.fillCircle(x1, y1, 2, WHITE);

  P(-v->dx, -v->dyR, x1, y1);
  P(v->dx, -v->dyF, x2, y2);

  if (v->shape == 6) { // Tank Tracks
    for (int i = 0; i <= 3; i++) {
      float lerp = i / 3.0f;
      int wx = x1 + (int)((x2 - x1) * lerp);
      int wy = y1 + (int)((y2 - y1) * lerp);
      drawWheelAt(wx, wy, (int)v->wrR, wa);
    }
    float dx_w = x2 - x1, dy_w = y2 - y1;
    float len = sqrtf(dx_w * dx_w + dy_w * dy_w);
    if (len > 0.1f) {
      float nx = -dy_w / len, ny = dx_w / len;
      int ox = (int)(nx * (v->wrR + 1)), oy = (int)(ny * (v->wrR + 1));
      display.drawLine(x1 + ox, y1 + oy, x2 + ox, y2 + oy, WHITE); 
      display.drawLine(x1 - ox, y1 - oy, x2 - ox, y2 - oy, WHITE); 
    }
  } else {
    drawWheelAt(x1, y1, (int)v->wrR, wa);
    drawWheelAt(x2, y2, (int)v->wrF, wa * v->wrR / v->wrF);
  }
}

enum { M_RESUME, M_NEW, M_VEH, M_STG, M_SND, M_RESTART, M_MAIN, M_RETRY, M_NONE = 255 };

uint8_t menuStep(const struct MenuItem* it, int n, int y0, bool press) {
  if (menuSel >= n) menuSel = 0;
  if (navY != 0) { menuSel = (menuSel + navY + n) % n; sfx(1500, 15); }
  display.setTextSize(1);
  for (int i = 0; i < n; i++) {
    int y = y0 + i * 11;
    if (i == menuSel) {
      display.fillRect(4, y - 2, 120, 11, WHITE);
      display.setTextColor(BLACK);
    } else {
      display.setTextColor(WHITE);
    }
    display.setCursor(12, y);
    display.print(it[i].label);
  }
  display.setTextColor(WHITE);
  if (press) { sfx(2000, 30); return it[menuSel].id; }
  return M_NONE;
}

void toggleSound() {
  soundOn = !soundOn;
  prefs.putBool("snd", soundOn);

}

void setup() {
  Serial.begin(115200);
  pinMode(PIN_SW, INPUT_PULLUP);
  analogReadResolution(12);
  buzzInit();

  Wire.begin(PIN_SDA, PIN_SCL);
  Wire.setClock(I2C_SPEED);
  if (!display.begin(SSD1306_SWITCHCAPVCC, OLED_ADDR)) {
    Serial.println("SSD1306 not found - check wiring / address");
    for (;;) delay(1000);
  }
  display.clearDisplay();
  display.display();

  delay(100);
  long sx = 0, sy = 0;
  for (int i = 0; i < 32; i++) { sx += analogRead(PIN_VRX); sy += analogRead(PIN_VRY); delay(2); }
  cxCal = sx / 32; cyCal = sy / 32;

  prefs.begin("hc32", false);
  for (int i=0; i<NSTAGES; i++) {
    char key[4]; snprintf(key, sizeof(key), "b%d", i);
    bestDist[i] = prefs.getUInt(key, 0);
  }
  
  vehIdx   = prefs.getUChar("veh", 0);
  stageIdx = prefs.getUChar("stg", 0);
  if (vehIdx < 0 || vehIdx >= NVEH) vehIdx = 0;
  if (stageIdx < 0 || stageIdx >= NSTAGES) stageIdx = 0;
  
  soundOn = prefs.getBool("snd", true);
  V = &vehicles[vehIdx];
  lastRunVeh = vehIdx;
  lastRunStage = stageIdx;

  resetGame();
  lastUs = micros();
}

void loop() {
  unsigned long now = micros();
  float dt = (now - lastUs) / 1000000.0f;
  lastUs = now;
  if (dt > 0.05f) dt = 0.05f;
  if (dt <= 0) dt = 0.001f;

  bool press = buttonEdge();
  readInputs();

  if (state != ST_PLAY && state != ST_OVER && millis() > sfxEnd) buzz(0);

  switch (state) {
    case ST_MENU:    runMainMenu(press);     break;
    case ST_VEHICLE: runVehicleSelect(press); break;
    case ST_STAGE:   runStageSelect(press);   break;
    case ST_PAUSE:   runPauseMenu(press);    break;
    case ST_OVER:    runGameOver(press);     break;

    case ST_PLAY: {
      if (press) { state = ST_PAUSE; menuSel = 0; buzz(0); break; }

      int steps = (int)ceilf(dt / 0.004f);
      if (steps < 1) steps = 1;
      float h = dt / steps;
      for (int i = 0; i < steps; i++) physicsStep(h);

      float g = fuel > 0 ? gasIn : 0;
      fuel -= (1.5f + 2.0f * g) * dt;
      if (fuel < 0) fuel = 0;

      pickups();

      float m = (car.x - 20.0f) / 8.0f;
      if (m > distM) distM = m;

      float tx = car.x - 36.0f + car.vx * 0.12f;
      camX += (tx - camX) * fminf(1.0f, 6.0f * dt);
      camY += (car.y - camY) * fminf(1.0f, 3.0f * dt);

      drawScene();
      display.display();

      if (millis() > sfxEnd) {
        if (g > 0.05f) buzz(70 + (int)(fabsf(car.vx) * 0.9f) + (int)(g * 40));
        else buzz(0);
      }

      if (crashed()) { gameOver("CRASHED!"); break; }
      if (fuel <= 0 && fabsf(car.vx) < 4.0f) {
        stuckT += dt;
        if (stuckT > 1.5f) gameOver("OUT OF FUEL");
      } else stuckT = 0;
      break;
    }
  }
}
